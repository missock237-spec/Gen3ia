// ============================================================
// WORKFLOW AUTHORIZATIONS (Comptes connectes aux agents)
// ============================================================

model WorkflowAuthorization {
  id           String    @id @default(cuid())
  userId       String
  service      String
  accessToken  String
  refreshToken String?
  accountId    String
  accountName  String
  scopes       String    @default("[]")
  isActive     Boolean   @default(true)
  lastUsedAt   DateTime?
  expiresAt    DateTime?
  createdAt    DateTime  @default(now())
  updatedAt    DateTime  @updatedAt

  user User @relation(fields: [userId], references: [id], onDelete: Cascade)

  @@unique([userId, service, accountId])
  @@index([userId])
  @@index([service])
  @@map("workflow_authorizations")
}
